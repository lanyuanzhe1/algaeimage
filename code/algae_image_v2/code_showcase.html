Step 1 — 结构张量偏振模拟

📄 code/algae_image_v2/core_engine/polarization_sim.py
使用 结构张量梯度法 从单张 RGB 明场图模拟四个偏振角度 (I₀/I₄₅/I₉₀/I₁₃₅)。
核心公式：梯度方向 → AoP，梯度幅值 → DoLP，再通过 Malus 定律 反推四个通道。
V1 算法原型位于 code/Polar_sim_0520/image_processing/polarization_sim.py 
def compute_polarization_sim(image_rgb: np.ndarray) -> np.ndarray:
    """从单张RGB明场图模拟四通道偏振图像 (I0, I45, I90, I135)"""
    gray = cv2.cvtColor(image_rgb, cv2.COLOR_RGB2GRAY).astype(np.float32)
    # 结构张量 — 梯度幅值 & 梯度方向
    Ix = cv2.Sobel(gray, cv2.CV_32F, 1, 0, ksize=3)
    Iy = cv2.Sobel(gray, cv2.CV_32F, 0, 1, ksize=3)
    G = np.sqrt(Ix**2 + Iy**2)
    theta = np.arctan2(Iy, Ix)
    # AoP ∈ [0, π), DoLP ∈ [0, 1]
    AoP = np.mod(theta, np.pi)
    DoLP = np.clip(G / (G.max() + 1e-8), 0, 1)
    # Malus 定律 → 四通道
    I0 = S0 * (1 + DoLP * np.cos(2 * AoP))
    I45 = S0 * (1 + DoLP * np.cos(2 * (AoP - np.pi/4)))
    I90 = S0 * (1 + DoLP * np.cos(2 * (AoP - np.pi/2)))
    I135 = S0 * (1 + DoLP * np.cos(2 * (AoP - 3*np.pi/4)))
    return np.stack([I0, I45, I90, I135], axis=2)

Step 2 — RDN 残差稠密网络偏振重建

📄 code/algae_image_v2/core_engine/reconstructor.py
RDBN (Residual Dense Network)：4通道输入 → 4通道输出，12个 RDB blocks，每 block 6层卷积。
参数量 ~0.6M，PSNR 62.46dB @ epoch 97。权重：weights/rdn_polarization.pth 
class RDB(nn.Module):
    """Residual Dense Block — 6层卷积 + 残差连接"""
    def __init__(self, nFeat=16, nLayer=6):
        for i in range(nLayer):
            self.conv.append(nn.Conv2d(nFeat*(i+1), nFeat, 3, 1, 1))

class PolarizationReconstructor(nn.Module):
    """RDN: 4→16→16→4, 12 RDB blocks"""
    def forward(self, x):
        # x: [B, 4, H, W] — I0/I45/I90/I135
        f = self.sfe(x)         # Shallow Feature Extraction
        for rdb in self.rdbs:    # 12× RDB blocks
            f = rdb(f)
        f = self.gff(f)         # Global Feature Fusion
        return self.recon(f)    # → [B, 4, H, W]

Step 3 — I_enh v2 去散射增强

📄 code/algae_image_v2/core_engine/enhancement.py
融合 Stokes 参数的去散射增强公式，通过 DoLP 加权 抑制散射背景、增强目标对比度。
核心公式在 enhance_image() 函数中。 
def compute_stokes(I0, I45, I90, I135):
    """计算 Stokes 参数 (S0, S1, S2) + DoLP + AoP"""
    S0 = I0 + I90
    S1 = I0 - I90
    S2 = I45 - I135
    DoLP = np.sqrt(S1**2 + S2**2) / (S0 + 1e-8)
    AoP = 0.5 * np.arctan2(S2, S1)
    return S0, S1, S2, DoLP, AoP

def enhance_image(S0, DoLP, AoP, alpha=1.0, gamma=0.5, beta=0.3):
    """I_enh v2: S0*(1+α-γ·DoLP+β·|sin(2·AoP)|·DoLP)"""
    I_enh = S0 * (1 + alpha - gamma * DoLP + beta * np.abs(np.sin(2 * AoP)) * DoLP)
    return np.clip(I_enh, 0, 255).astype(np.uint8)

Step 4 — YOLOv8s 藻种检测

📄 code/algae_image_v2/core_engine/inference.py
YOLOv8s 检测 5 类 FMPD：Woronichinia / Spiroides / Dinobryon / Other-phytoplankton / Non-phytoplankton。
默认模型 mAP50 73.9%，权重：weights/best_v8s.pt (~22MB) 
class AlgaeDetector:
    def __init__(self, weights, device='cuda', imgsz=640):
        self.model = YOLO(weights)
        self.class_names = [
            'Woronichinia', 'Spiroides', 'Dinobryon',
            'Other-phytoplankton', 'Non-phytoplankton']

    def detect(self, image):
        results = self.model(image, imgsz=self.imgsz, conf=0.25, iou=0.7)
        return [{
            'class': self.class_names[int(b.cls)], 'conf': float(b.conf),
            'bbox': b.xyxy.tolist()
        } for b in results[0].boxes]
